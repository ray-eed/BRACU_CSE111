import java.util.Scanner;

public class HW1 {
   public static boolean isPrime(int n) {
       if (n < 2){
       return false;
       }
       for (int j = 2; j * j <= n; j++) {
           if (n % j == 0)
               return false;
       }
       return true;
   }
   public static void main(String[] args) {
       Scanner sc = new Scanner(System.in);
           System.out.print("Enter the first number: ");
           int x=sc.nextInt();
           System.out.print("Enter the second number: ");
           int y=sc.nextInt();
           int temp;
           if(x>y){
               temp=x;
               x=y;
               y=temp;
           }
           int count=0;
           for(int i=x;i<=y;i++){
               if(isPrime(i)){
                   count++;
               }
       }
       System.out.println("There are "+count+" prime numbers between "+x+" and "+y);
       sc.close();
   }
}

       for(int i=0;i<x.length();i++){

import java.util.Scanner;

public class HW3 {
   public static void main(String[] args) {
       Scanner sc = new Scanner(System.in);
       System.out.print("N = ");
       int x = sc.nextInt();
       int[] arr = new int[x];
       for(int i=0;i<x;i++){
           int l=sc.nextInt();
           arr[i]=l;
       }
       boolean[] counted = new boolean[x];
       for(int j=0;j<x;j++){
           if(!counted[j]){
               int count=1;
               for(int k=j+1;k<x;k++){
                   if(arr[j]==arr[k]){
                       count++;
                       counted[k]=true;
                   }
               }
               System.out.println(arr[j]+" - "+count+" times");
           }
       }
       sc.close();
   }
}
